  input  : stage_01/01_ais_clean.csv
  output : stage_02/02_vessel_interpolated_grid.csv

  input  : stage_02/02_vessel_interpolated_grid.csv
  output : stage_03/03_input_state_enhanced.csv

04_No_Intervention_Forecast.ipynb

           <DRIVE_ROOT>/data_raw/vehicle_arrival_rate_30min.csv
  output : stage_04/04_fuzzy_input.csv,
           stage_04/04_daily_port_queue_forecast.csv,
           stage_04/04_daily_event_log.csv

05_Fuzzification.ipynb
  input  : stage_04/04_fuzzy_input.csv
  output : stage_05/05_fuzzy_memberships.csv

06_Rule_Evaluation.ipynb
  input  : stage_05/05_fuzzy_memberships.csv
  output : stage_06/06_rule_evaluation.csv

07_Candidate_Action.ipynb

  output : seluruh artefak 07_* di stage_07
